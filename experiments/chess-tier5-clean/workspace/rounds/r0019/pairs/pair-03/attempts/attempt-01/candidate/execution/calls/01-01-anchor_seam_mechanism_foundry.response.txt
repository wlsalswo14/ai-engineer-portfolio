```python
class ForcingObligationGate:
    REQUIRED = (
        "is_legal", "in_check", "gives_check", "is_promotion",
        "is_capture", "is_recapture", "captured_value",
        "attacker_value", "depth", "extension_budget",
        "extension_streak",
    )

    def __init__(self):
        self.begin_count = 0
        self.calls = 0
        self.work = 0

    def begin(self):
        self.begin_count += 1
        self.calls = 0
        self.work = 0

    def step(self, state):
        missing = [k for k in self.REQUIRED if k not in state]
        if missing:
            raise ValueError("missing dependency: " + ",".join(missing))

        self.calls += 1

        if not state["is_legal"]:
            out = {
                "extension": 0, "move_order_bonus": 0,
                "tactical": False, "work_units": 1,
                "tags": ["illegal"],
            }
            self.work += 1
            return out

        tags = []
        if state["in_check"]: tags.append("check")
        if state["gives_check"]: tags.append("check")
        if state["is_promotion"]: tags.append("promotion")
        if state["is_capture"]: tags.append("capture")
        if state["is_recapture"]: tags.append("recapture")

        high_capture = (
            state["is_capture"] and
            state["captured_value"] >= state["attacker_value"]
        )
        if high_capture:
            tags.append("hvcapture")

        bonus = 0
        if state["in_check"]: bonus += 400
        if state["gives_check"]: bonus += 300
        if state["is_promotion"]: bonus += 250
        if state["is_capture"]:
            bonus += 100 + min(
                900,
                max(0, state["captured_value"] - state["attacker_value"]),
            )
        if state["is_recapture"]: bonus += 150

        hard = bool(
            state["in_check"] or state["gives_check"] or
            state["is_promotion"] or high_capture or
            state["is_recapture"]
        )
        eligible = (
            state["depth"] <= 2 and
            state["extension_budget"] > 0 and
            state["extension_streak"] < 2
        )
        extension = 1 if hard and eligible else 0

        out = {
            "extension": extension,
            "move_order_bonus": bonus,
            "tactical": hard,
            "work_units": 1 + extension,
            "tags": tags,
        }
        self.work += out["work_units"]
        return out
```

```python
class AnchorSeamSurrogate:
    def __init__(self, gate):
        self.gate = gate
        self.begin()

    def begin(self, depth=2, extension_budget=2,
              extension_streak=0, last_to=-1):
        self.gate.begin()
        self.node = {
            "depth": depth,
            "extension_budget": extension_budget,
            "extension_streak": extension_streak,
            "last_to": last_to,
        }

    def apply(self, facts, move_to):
        event = dict(facts)
        event.update(self.node)
        decision = self.gate.step(event)

        next_node = {
            "depth": self.node["depth"] - 1 + decision["extension"],
            "extension_budget": (
                self.node["extension_budget"] - decision["extension"]
            ),
            "extension_streak": (
                self.node["extension_streak"] + 1
                if decision["extension"] else 0
            ),
            "last_to": move_to,
        }
        self.node = next_node
        return decision, next_node
```
