def advance(state, event):
    if event.ply != state.last_ply + 1:
        raise ValueError("invalid capability transition")
    if event.target not in TARGET_CLASSES:
        raise ValueError("invalid capability target")

    quiet = not event.capture and not event.promotion and not event.gives_check
    trigger = quiet and event.target != "none"

    if trigger and event.depth_left <= 1 and state.credits:
        return Transition(
            CapabilityState(state.credits - 1, event.ply), 1
        )
    if quiet:
        return Transition(CapabilityState(MAX_CREDITS, event.ply), 0)
    return Transition(CapabilityState(state.credits, event.ply), 0)
