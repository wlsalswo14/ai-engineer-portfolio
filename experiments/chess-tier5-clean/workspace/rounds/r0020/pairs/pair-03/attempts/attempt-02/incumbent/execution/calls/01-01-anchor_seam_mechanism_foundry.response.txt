## Compact evidence artifact

Capability: `RepetitionLedger` — reversible threefold-repetition detection keyed by `(board, side, castling, en-passant)`. It is task-general, independent of evaluation and move ordering.

### Component source

```python
class RepetitionLedger:
    def __init__(self, identity):
        if not callable(identity):
            raise TypeError("identity must be callable")
        self.identity = identity
        self.counts = {}
        self.stack = []

    def begin(self, states=()):
        self.counts.clear()
        self.stack.clear()
        for state in states:
            self.push(state)

    def push(self, state):
        key = self.identity(state)
        prior = self.counts.get(key, 0)
        self.counts[key] = prior + 1
        self.stack.append((key, prior))
        return len(self.stack)

    def pop(self, token=None):
        if not self.stack:
            raise IndexError("empty repetition path")
        if token is not None and token != len(self.stack):
            raise RuntimeError("non-LIFO repetition pop")
        key, prior = self.stack.pop()
        if prior:
            self.counts[key] = prior
        else:
            del self.counts[key]

    def count(self, state):
        return self.counts.get(self.identity(state), 0)

    def is_threefold(self, state):
        return self.count(state) >= 3

    def __len__(self):
        return len(self.stack)
```

### Anchor seam surrogate

Provenance: the anchor’s `Position.__init__` fields (`b`, `side`, `cast`, `ep`, `half`, `full`), `make()` transition, `setpos()` replay lifecycle, and recursive `search()`/`qsearch()` consumers.

```python
from repetition_ledger import RepetitionLedger

class AnchorState:
    __slots__ = ("b", "side", "cast", "ep", "half", "full")

    def __init__(self, board, side=0, cast=0, ep=-1, half=0, full=1):
        self.b = list(board)
        self.side, self.cast, self.ep = side, cast, ep
        self.half, self.full = half, full

def identity(p):
    return (tuple(p.b), p.side, p.cast, p.ep)

def make(p, move):
    a, z = move
    b = p.b[:]
    piece, captured = b[a], b[z]
    b[a], b[z] = ".", piece
    return AnchorState(
        b, p.side ^ 1, p.cast, p.ep,
        0 if piece.lower() == "p" or captured != "." else p.half + 1,
        p.full + (1 if p.side == 0 else 0),
    )

def trace(states, key):
    r = RepetitionLedger(key)
    root = states[0]
    r.begin((root,))
    out = []
    for s in states[1:]:
        t = r.push(s)
        out.append((r.count(root), r.is_threefold(root), t))
    return out, r

# One reversible four-transition cycle, repeated twice.
board = ["."] * 64
board[0], board[7] = "R", "r"
root = AnchorState(board)
a = make(root, (0, 1))
b = make(a, (7, 6))
c = make(b, (1, 0))
root2 = make(c, (6, 7))
a2 = make(root2, (0, 1))
b2 = make(a2, (7, 6))
c2 = make(b2, (1, 0))
root3 = make(c2, (6, 7))

states = (root, a, b, c, root2, a2, b2, c2, root3)
seam_trace, seam_ledger = trace(states, identity)
primitive_trace, _ = trace(tuple(identity(s) for s in states), lambda x: x)

assert [(n, d) for n, d, _ in seam_trace] == \
       [(n, d) for n, d, _ in primitive_trace]
assert seam_trace[-1][:2] == (3, True)

null = RepetitionLedger(identity)
null.begin((root,))
null.push(a)
assert not null.is_threefold(root)

life = RepetitionLedger(identity)
life.begin((root,))
token = life.push(a)
before = (life.count(a), len(life))
life.pop(token)
after = (life.count(a), len(life))
reuse = life.push(a)
assert (before, after, (life.count(a), len(life), reuse)) == \
       ((1, 2), (0, 1), (1, 2, 2))

closure = RepetitionLedger(identity)
closure.begin((root,))
assert closure.count(AnchorState(root.b, root.side, root.cast, root.ep, 99, 40)) == 1

bounded = RepetitionLedger(identity)
bounded.begin((root,))
for i in range(256):
    bounded.push(a if i & 1 else root2)
max_depth = len(bounded)
while bounded:
    bounded.pop()
assert (max_depth, len(bounded)) == (257, 0)

print("primitive_effect=", [(n, d) for n, d, _ in primitive_trace])
print("seam_effect=", [(n, d) for n, d, _ in seam_trace])
print("null=", (null.count(root), null.is_threefold(root)))
print("lifecycle=", before, after, (life.count(a), len(life), reuse))
print("half_full_ignored=", closure.count(root))
print("bounded_work=", max_depth, len(bounded))
```
