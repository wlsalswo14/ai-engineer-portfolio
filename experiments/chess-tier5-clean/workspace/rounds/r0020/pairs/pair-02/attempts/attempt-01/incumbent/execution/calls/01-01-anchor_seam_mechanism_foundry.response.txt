### Component source

```python
class RepetitionLedger:
    """Primitive: enter(key), leave(), count(key), is_draw(key), reset()."""

    THRESHOLD = 3

    def __init__(self):
        self._counts = {}
        self._stack = []

    def reset(self):
        self._counts.clear()
        self._stack.clear()

    def enter(self, key):
        if not isinstance(key, tuple) or len(key) != 4:
            raise TypeError("key must be (board, side, castling, ep)")
        self._stack.append(key)
        n = self._counts.get(key, 0) + 1
        self._counts[key] = n
        return n

    def leave(self):
        if not self._stack:
            raise RuntimeError("unbalanced leave")
        key = self._stack.pop()
        n = self._counts[key] - 1
        if n == 0:
            del self._counts[key]
        else:
            self._counts[key] = n
        return key

    def count(self, key):
        return self._counts.get(key, 0)

    def is_draw(self, key):
        return self.count(key) >= self.THRESHOLD

    def snapshot(self):
        return len(self._stack), dict(self._counts)
```

```python
from dataclasses import dataclass, replace

@dataclass(frozen=True)
class AnchorState:
    b: tuple
    side: int
    cast: int
    ep: int
    half: int
    full: int

    def seam_key(self):
        return (''.join(self.b), self.side, self.cast, self.ep)

    def next(self, **changes):
        return replace(self, **changes)


class AnchorSearchSeam:
    def __init__(self, ledger):
        self.ledger = ledger
        self.calls = 0
        self.max_depth = 0

    def run(self, states):
        self.calls = 0
        self.max_depth = 0

        def visit(index, depth):
            state = states[index]
            self.calls += 1
            self.max_depth = max(self.max_depth, depth)
            key = state.seam_key()
            occurrence = self.ledger.enter(key)
            try:
                if occurrence >= self.ledger.THRESHOLD:
                    return 0
                if index + 1 == len(states):
                    return 1
                return visit(index + 1, depth + 1)
            finally:
                self.ledger.leave()

        return visit(0, 1)
```
