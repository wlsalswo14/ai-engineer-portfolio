```python
class RepetitionLedger:
    def __init__(self, key_fn, limit=3):
        self.key_fn = key_fn
        self.limit = limit
        self.counts = {}
        self.path = []

    def reset(self):
        self.counts.clear()
        self.path.clear()

    def enter(self, state):
        key = self.key_fn(state)
        self.path.append(key)
        self.counts[key] = self.counts.get(key, 0) + 1
        return self.counts[key]

    def leave(self):
        if not self.path:
            raise RuntimeError("unbalanced repetition lifecycle")
        key = self.path.pop()
        n = self.counts[key] - 1
        if n:
            self.counts[key] = n
        else:
            del self.counts[key]

    def terminal(self, state):
        return self.counts.get(self.key_fn(state), 0) >= self.limit

    def depth(self):
        return len(self.path)
```

```python
def anchor_key(p):
    return (tuple(p.b), p.side, p.cast, p.ep)

class AnchorSearchSeam:
    def __init__(self, max_steps=64):
        self.ledger = RepetitionLedger(anchor_key)
        self.max_steps = max_steps

    def run(self, root, moves, legal, make, notation):
        self.ledger.reset()
        self.ledger.enter(root)
        p = root
        trace = []

        for ply, uci in enumerate(moves, 1):
            if ply > self.max_steps:
                raise RuntimeError("bounded seam work exceeded")
            move = next(m for m in legal(p) if notation(m) == uci)
            p = make(p, move)
            occurrence = self.ledger.enter(p)
            draw = self.ledger.terminal(p)
            trace.append((ply, uci, occurrence, p.side,
                          p.cast, p.ep, p.half, p.full, draw))
            if draw:
                break

        result = 0 if self.ledger.terminal(p) else None
        depth = self.ledger.depth()
        while self.ledger.depth():
            self.ledger.leave()
        return result, trace, depth <= self.max_steps
```
