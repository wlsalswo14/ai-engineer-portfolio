import json
from dataclasses import dataclass


def _read(position, *names):
    for name in names:
        if hasattr(position, name):
            return getattr(position, name)
    raise AttributeError("position lacks " + "/".join(names))


def canonical_key(position, ep_relevant=None):
    board = tuple(_read(position, "b", "board"))
    if len(board) != 64:
        raise ValueError("board must contain 64 squares")
    side = int(_read(position, "side", "turn"))
    castling = int(_read(position, "cast", "castling"))
    ep = int(_read(position, "ep", "en_passant"))
    if ep >= 0 and ep_relevant is not None and not ep_relevant(position):
        ep = -1
    return board, side, castling, ep


class RepetitionLedger:
    """Game-history counts plus a reversible search-path overlay."""

    def __init__(self, ep_relevant=None):
        self.ep_relevant = ep_relevant
        self._counts = {}
        self._path = []

    def reset(self):
        if self._path:
            raise RuntimeError("cannot reset during an active search path")
        self._counts.clear()

    def record(self, position):
        if self._path:
            raise RuntimeError("game-history record overlaps search path")
        key = canonical_key(position, self.ep_relevant)
        self._counts[key] = self._counts.get(key, 0) + 1
        return self._counts[key]

    def record_many(self, positions):
        last = 0
        for position in positions:
            last = self.record(position)
        return last

    def count(self, position):
        return self._counts.get(
            canonical_key(position, self.ep_relevant), 0
        )

    def is_threefold(self, position):
        return self.count(position) >= 3

    def enter(self, position):
        mark = len(self._path)
        self.push(position)
        return mark

    def push(self, position):
        key = canonical_key(position, self.ep_relevant)
        self._path.append(key)
        self._counts[key] = self._counts.get(key, 0) + 1
        return self._counts[key]

    def pop(self):
        if not self._path:
            raise IndexError("empty search path")
        key = self._path.pop()
        count = self._counts[key] - 1
        if count:
            self._counts[key] = count
        else:
            del self._counts[key]
        return count

    def leave(self, mark):
        if not isinstance(mark, int) or mark < 0 or mark > len(self._path):
            raise ValueError("invalid search-path mark")
        while len(self._path) > mark:
            self.pop()

    def search_guard(self, position, draw_score=0):
        return draw_score if self.is_threefold(position) else None
