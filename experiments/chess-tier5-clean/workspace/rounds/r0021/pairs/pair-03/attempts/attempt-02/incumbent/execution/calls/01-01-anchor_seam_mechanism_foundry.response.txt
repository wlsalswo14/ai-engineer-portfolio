class PathRepetitionGuard:
    def __init__(self):
        self._counts = {}
        self._stack = []

    def enter(self, key):
        if self._counts.get(key, 0):
            return False
        self._counts[key] = 1
        self._stack.append(key)
        return True

    def leave(self, key):
        if not self._stack or self._stack[-1] != key:
            raise RuntimeError("non-LIFO repetition-guard leave")
        self._stack.pop()
        del self._counts[key]

    @property
    def depth(self):
        return len(self._stack)

class AnchorQsearchSeam:
    def __init__(self, guard, max_ply=12):
        self.guard = guard
        self.max_ply = max_ply
        self.trace = []
        self.nodes = 0
        self.max_active = 0

        self.trace = []
        self.nodes = 0
        self.max_active = 0

        def visit(state, ply):
            self.nodes += 1
            key = anchor_state_key(state)

            if not self.guard.enter(key):
                self.trace.append(("repeat-cut", ply))
                return "repeat"

            self.max_active = max(self.max_active, self.guard.depth)
            self.trace.append(("enter", ply, state.full))

            try:
                if ply >= self.max_ply:
                    self.trace.append(("bounded-cut", ply))
                    return "bound"

                children = tuple(forcing_children(state))
                self.trace.append(("consume", ply, len(children)))

                if not children:
                    self.trace.append(("terminal", ply))
                    return "terminal"

                return [visit(child, ply + 1) for child in children]
            finally:
                self.guard.leave(key)
                self.trace.append(("leave", ply, self.guard.depth))

        return visit(position, 0)
