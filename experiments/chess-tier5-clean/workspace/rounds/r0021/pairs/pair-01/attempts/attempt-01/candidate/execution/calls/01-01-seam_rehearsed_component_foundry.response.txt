import json
import sys

IDENTITY = "bounded-late-quiet-research-v1"
MAX_LEGAL_MOVES = 256


def initial():
    return {"epoch": 0, "open": False}


def _int(value, name, low, high):
    if type(value) is not int or value < low or value > high:
        raise ValueError("invalid " + name)
    return value


def _bool(value, name):
    if type(value) is not bool:
        raise TypeError("invalid " + name)
    return value


def begin_node(state, depth_left, ply, legal_count):
    if not isinstance(state, dict) or state.get("open") is not False:
        raise ValueError("node already open")
    if type(state.get("epoch")) is not int or state["epoch"] < 0:
        raise ValueError("invalid lifecycle state")

    depth_left = _int(depth_left, "depth_left", 0, 128)
    ply = _int(ply, "ply", 0, 256)
    legal_count = _int(legal_count, "legal_count", 1, MAX_LEGAL_MOVES)

    return {
        "epoch": state["epoch"] + 1,
        "open": True,
        "depth_left": depth_left,
        "ply": ply,
        "legal_count": legal_count,
        "next_index": 0,
        "visits": 0,
    }


def visit_node(state, event):
    if not isinstance(state, dict) or state.get("open") is not True:
        raise ValueError("node is closed")
    if not isinstance(event, dict):
        raise TypeError("event must be an object")

    index = _int(event.get("move_index"), "move_index", 0, MAX_LEGAL_MOVES - 1)
    if index != state["next_index"] or index >= state["legal_count"]:
        raise ValueError("move order")

    names = (
        "capture",
        "promotion",
        "gives_check",
        "pv_move",
        "tt_move",
        "killer_move",
    )
    for name in names:
        _bool(event.get(name), name)

    quiet = not (
        event["capture"] or event["promotion"] or event["gives_check"]
    )
    eligible = (
        state["depth_left"] >= 3
        and index >= 3
        and state["legal_count"] >= 8
        and quiet
        and not event["pv_move"]
        and not event["tt_move"]
        and not event["killer_move"]
    )

    reduction = 0
    if eligible:
        reduction = 1
        if state["depth_left"] >= 6 and index >= 8:
            reduction += 1
        reduction = min(reduction, state["depth_left"] - 1)

    decision = {
        "full_depth": state["depth_left"],
        "probe_depth": state["depth_left"] - reduction,
        "reduction": reduction,
        "reduced": reduction > 0,
        "requires_full_research": reduction > 0,
        "reason": "late_quiet" if reduction else "full_depth",
    }

    next_state = dict(state)
    next_state["next_index"] += 1
    next_state["visits"] += 1
    return next_state, decision


def end_node(state):
    if not isinstance(state, dict) or state.get("open") is not True:
        raise ValueError("node is closed")
    return {
        "epoch": state["epoch"],
        "open": False,
        "visits": state["visits"],
    }


def dispatch(request):
    operation = request.get("op")
    if operation == "initial":
        return initial()
    if operation == "begin":
        return begin_node(
            request["state"],
            request["depth_left"],
            request["ply"],
            request["legal_count"],
        )
    if operation == "visit":
        state, decision = visit_node(request["state"], request["event"])
        return {"state": state, "decision": decision}
    if operation == "end":
        return end_node(request["state"])
    raise ValueError("unknown operation")


def main():
    for line in sys.stdin:
        if not line.strip():
            continue
        try:
            result = dispatch(json.loads(line))
            print(json.dumps({"ok": True, "result": result}, separators=(",", ":")))
        except Exception as exc:
            print(json.dumps({"ok": False, "error": str(exc)}, separators=(",", ":")))
        sys.stdout.flush()


if __name__ == "__main__":
    main()
