from collections import OrderedDict
from dataclasses import dataclass

@dataclass(frozen=True)
class Entry:
    depth: int
    score: int
    bound: str
    best: object

class BoundedTT:
    def __init__(self, limit=64):
        self.limit = max(1, int(limit))
        self.table = OrderedDict()
        self.generation = 0
        self.last_work = 0

    def new_game(self):
        self.generation += 1
        self.table.clear()
        self.last_work = 1

    def save(self, key, depth, score, alpha, beta, best):
        bound = "U" if score <= alpha else "L" if score >= beta else "E"
        self.table.pop(key, None)
        self.table[key] = Entry(depth, score, bound, best)
        self.last_work = 1
        if len(self.table) > self.limit:
            self.table.popitem(last=False)
            self.last_work = 2
        return bound

    def probe(self, key, depth, alpha, beta):
        self.last_work = 1
        entry = self.table.get(key)
        if entry is None:
            return False, None, None
        self.table.move_to_end(key)
        if entry.depth < depth:
            return False, None, entry.best
        if entry.bound == "E":
            return True, entry.score, entry.best
        if entry.bound == "L" and entry.score >= beta:
            return True, entry.score, entry.best
        if entry.bound == "U" and entry.score <= alpha:
            return True, entry.score, entry.best
        return False, None, entry.best


class AnchorTTSeam:
    """Surrogate for the incumbent's Engine.search <-> self.tt seam."""
    REQUIRED = ("board", "side", "cast", "ep", "half", "full")

    def __init__(self, cache):
        self.cache = cache
        self.state = None

    def position(self, state):
        if any(k not in state for k in self.REQUIRED):
            raise ValueError("incomplete anchor state")
        board = state["board"]
        if not isinstance(board, str):
            board = "".join(board)
        if len(board) != 64:
            raise ValueError("anchor board shape")
        self.state = dict(state)
        self.state["board"] = board

    def key(self):
        if self.state is None:
            raise RuntimeError("no position")
        return (
            self.state["board"],
            self.state["side"],
            self.state["cast"],
            self.state["ep"],
        )

    def produce(self, depth, score, alpha, beta, best):
        return self.cache.save(self.key(), depth, score, alpha, beta, best)

    def consume(self, depth, alpha, beta):
        return self.cache.probe(self.key(), depth, alpha, beta)

    def new_game(self):
        self.cache.new_game()
        self.state = None


BOARD = "rnbqkbnrpppppppp................................PPPPPPPPRNBQKBNR"
