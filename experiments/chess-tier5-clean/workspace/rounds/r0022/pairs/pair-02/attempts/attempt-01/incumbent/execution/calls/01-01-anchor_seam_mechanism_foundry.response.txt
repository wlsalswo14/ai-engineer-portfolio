import json

class CycleGuard:
    def __init__(self, max_ply=128):
        if not isinstance(max_ply, int) or max_ply < 1:
            raise ValueError("max_ply must be positive")
        self.max_ply = max_ply
        self._stack = []
        self._counts = {}
        self.operations = 0

    def reset(self):
        self._stack.clear()
        self._counts.clear()
        self.operations = 0

    def push(self, key):
        hash(key)
        if len(self._stack) >= self.max_ply:
            raise OverflowError("cycle-guard path bound exceeded")
        count = self._counts.get(key, 0) + 1
        self._stack.append(key)
        self._counts[key] = count
        self.operations += 1
        return {
            "depth": len(self._stack),
            "occurrences": count,
            "draw": count >= 3,
        }

    def pop(self):
        if not self._stack:
            raise RuntimeError("cycle-guard underflow")
        key = self._stack.pop()
        count = self._counts[key] - 1
        if count:
            self._counts[key] = count
        else:
            del self._counts[key]
        self.operations += 1
        return {"depth": len(self._stack), "key": key}

    def empty(self):
        return not self._stack and not self._counts


class AnchorSeamSurrogate:
    def __init__(self, guard, max_steps=128):
        self.guard = guard
        self.max_steps = max_steps

    def consume_path(self, root_key, child_keys):
        if len(child_keys) + 1 > self.max_steps:
            raise OverflowError("surrogate work bound exceeded")

        trace = []
        self.guard.reset()

        def consume(key, tail):
            state = self.guard.push(key)
            trace.append(
                ("enter", state["depth"], state["occurrences"], state["draw"])
            )
            try:
                if state["draw"]:
                    trace.append(("consumer", "draw", 0))
                    return 0
                if not tail:
                    trace.append(("consumer", "leaf", 1))
                    return 1
                return consume(tail[0], tail[1:])
            finally:
                trace.append(("leave", self.guard.pop()["depth"]))

        try:
            result = consume(root_key, list(child_keys))
            trace.append(("closed", self.guard.empty(), self.guard.operations))
            return result, trace
        finally:
            self.guard.reset()


def key(tag, side=0, cast=0, ep=-1):
    return (tag * 64, side, cast, ep)
