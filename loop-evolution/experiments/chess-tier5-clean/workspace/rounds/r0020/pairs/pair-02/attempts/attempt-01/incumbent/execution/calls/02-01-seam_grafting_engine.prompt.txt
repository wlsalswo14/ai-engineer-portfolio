class RepetitionLedger:
    """Primitive: enter(key), leave(), count(key), is_draw(key), reset()."""

    THRESHOLD = 3

    def __init__(self):
        self._counts = {}
        self._stack = []

    def reset(self):
        self._counts.clear()
        self._stack.clear()

    def enter(self, key):
        if not isinstance(key, tuple) or len(key) != 4:
            raise TypeError("key must be (board, side, castling, ep)")
        self._stack.append(key)
        n = self._counts.get(key, 0) + 1
        self._counts[key] = n
        return n

    def leave(self):
        if not self._stack:
            raise RuntimeError("unbalanced leave")
        key = self._stack.pop()
        n = self._counts[key] - 1
        if n == 0:
            del self._counts[key]
        else:
            self._counts[key] = n
        return key

    def count(self, key):
        return self._counts.get(key, 0)

    def is_draw(self, key):
        return self.count(key) >= self.THRESHOLD

    def snapshot(self):
        return len(self._stack), dict(self._counts)
