Artifact: one capability plus one anchor-seam surrogate; no engine emitted.

Capability: bounded frontier-threat extension gate. It mirrors the anchor’s `capability_advance` lifecycle: quiet newly-attacking moves at depth ≤1 consume one credit and extend one ply; quiet null moves reset credits; tactical moves preserve credits.

```python
from dataclasses import dataclass

TARGETS = frozenset(("none", "minor", "major"))
MAX_CREDITS = 2

@dataclass(frozen=True)
class State:
    credits: int
    last_ply: int

@dataclass(frozen=True)
class Event:
    ply: int
    depth_left: int
    capture: bool
    promotion: bool
    gives_check: bool
    target: str

class ThreatExtension:
    def initial(self):
        return State(MAX_CREDITS, -1)

    def advance(self, state, event):
        if event.ply != state.last_ply + 1:
            raise ValueError("non-contiguous ply")
        if event.depth_left < 0 or event.target not in TARGETS:
            raise ValueError("invalid event")

        quiet = not (event.capture or event.promotion or event.gives_check)
        trigger = quiet and event.target != "none"

        if trigger and event.depth_left <= 1 and state.credits:
            return State(state.credits - 1, event.ply), 1
        if quiet:
            return State(MAX_CREDITS, event.ply), 0
        return State(state.credits, event.ply), 0


@dataclass(frozen=True)
class Position:
    # Exact anchor Position shape.
    b: tuple
    side: int
    cast: int
    ep: int
    half: int
    full: int

@dataclass(frozen=True)
class Move:
    # Exact decoded fields consumed at the anchor seam.
    a: int
    z: int
    pr: int = 0
    fl: int = 0

@dataclass(frozen=True)
class SeamFrame:
    # q is the result of anchor make(p, m).
    p: Position
    q: Position
    m: Move
    depth_left: int
    ply: int
    king_square: int
    attacks: object

class AttackDependency:
    """Bounded surrogate for anchor attacked/newly_attacked dependencies."""
    def __init__(self, attack_sets):
        self.attack_sets = attack_sets
        self.calls = []

    def attacked(self, p, square, side):
        self.calls.append(("attacked", square, side))
        return square in self.attack_sets.get((p.b, side), frozenset())

    def newly_attacked(self, p, q, side):
        self.calls.append(("newly_attacked", side))
        result = "none"
        for square, piece in enumerate(q.b):
            enemy = piece != "." and piece.isupper() != (side == 0)
            if enemy and piece.lower() in "nbrq":
                if self.attacked(q, square, side) and not self.attacked(p, square, side):
                    if piece.lower() in "rq":
                        return "major"
                    result = "minor"
        return result

class AnchorSeamSurrogate:
    """Translates the real Engine.search seam into the primitive interface."""
    def __init__(self, capability):
        self.capability = capability

    def event(self, frame):
        p, q, m = frame.p, frame.q, frame.m
        capture = p.b[m.z] != "." or bool(m.fl & 2)
        promotion = bool(m.pr)
        gives_check = frame.attacks.attacked(q, frame.king_square, p.side)

        target = "none"
        if not capture and not promotion and not gives_check:
            target = frame.attacks.newly_attacked(p, q, p.side)

        return Event(
            frame.ply, frame.depth_left, capture,
            promotion, gives_check, target
        )

    def consume(self, state, frame):
        event = self.event(frame)
        next_state, extension = self.capability.advance(state, event)

        # Existing downstream obligation:
        # nd += extension; recurse with next_state.
        return next_state, extension, frame.depth_left + extension, event
```
