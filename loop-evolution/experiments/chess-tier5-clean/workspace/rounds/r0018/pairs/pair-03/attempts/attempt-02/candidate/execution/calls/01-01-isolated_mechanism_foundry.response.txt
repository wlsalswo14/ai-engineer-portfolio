import sys

INF = 10**9


class ForcingFrontier:
    def __init__(self, max_nodes=128, max_plies=8):

            raise ValueError("max_nodes must be positive")

        self.ply_limited = False
        self.trace = []
        score, action = self._visit(state, 0)
        return {
            "score": score,
            "best_action": action,
            "nodes": self.nodes,
            "node_limited": self.node_limited,
            "ply_limited": self.ply_limited,
            "trace": self.trace,
        }

    @classmethod
    def _validate(cls, state):
        if not isinstance(state, dict) or type(state.get("static")) is not int:
            raise TypeError("state.static must be an integer")
        if type(state.get("forced", False)) is not bool:
            raise TypeError("state.forced must be boolean")
        actions = state.get("actions", [])
        if not isinstance(actions, list):
            raise TypeError("state.actions must be a list")
        ids = set()
        for a in actions:
            if not isinstance(a, dict):
                raise TypeError("actions must be records")
            ident = a.get("id")
            if not isinstance(ident, str) or not ident or ident in ids:
                raise ValueError("action ids must be unique strings")
            ids.add(ident)
            if type(a.get("forcing")) is not bool:
                raise TypeError("action.forcing must be boolean")
            cls._validate(a["child"])

    def _visit(self, state, depth):
        tag = state.get("id", "?")
        if self.nodes >= self.max_nodes:
            self.node_limited = True
            self.trace.append({"state": tag, "depth": depth,
                               "event": "node-boundary"})
            return state["static"], None

        if depth >= self.max_plies:
            self.ply_limited = True
            self.trace.append({"state": tag, "depth": depth,

            return state["static"], None

        actions = [a for a in state.get("actions", []) if a["forcing"]]
        actions.sort(key=lambda a: a["id"])

        if not actions:
            self.trace.append({
                "state": tag, "depth": depth, "event": "frontier",
                "score": state["static"], "best_action": None
            })
            return state["static"], None

        best = -INF if state.get("forced", False) else state["static"]

                break
            child_score, _ = self._visit(action["child"], depth + 1)
            score = -child_score
            if best_action is None or score > best:
                best = score
                best_action = action["id"]

        if best_action is None:
            best = state["static"]

        self.trace.append({
            "state": tag, "depth": depth, "event": "forcing",
            "score": best, "best_action": best_action
        })
        return best, best_action
