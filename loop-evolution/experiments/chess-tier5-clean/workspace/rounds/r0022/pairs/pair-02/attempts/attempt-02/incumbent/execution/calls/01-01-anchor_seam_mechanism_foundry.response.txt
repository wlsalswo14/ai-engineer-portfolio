## Compact artifact

Capability: transactional root-incumbent continuity. It preserves the last fully completed iterative-deepening root result across timeout and orders the next root search around it.

```python
import json
import math


class RootIncumbent:
    """One bounded, transactional iterative-deepening root incumbent."""

    def __init__(self):
        self._state = None
        self._generation = 0
        self.last_work = 0

    def begin(self, position_key, legal_moves, fallback):
        moves = tuple(legal_moves)
        if self._state is not None and self._state["phase"] != "closed":
            raise RuntimeError("previous search is still active")
        if not moves or fallback not in moves:
            raise ValueError("fallback must be legal")
        self._generation += 1
        self._state = {
            "key": position_key,
            "moves": moves,
            "best": fallback,
            "depth": 0,
            "score": None,
            "phase": "active",
        }
        self.last_work = 1
        return ("begin", self._generation, position_key, fallback)

    def _active(self, key):
        if self._state is None or self._state["key"] != key:
            raise RuntimeError("inactive position")
        if self._state["phase"] == "closed":
            raise RuntimeError("search closed")
        return self._state

    def order(self, key, moves):
        state = self._active(key)
        if state["phase"] != "active":
            raise RuntimeError("root is not active")
        supplied = tuple(moves)
        if len(supplied) != len(state["moves"]) or set(supplied) != set(state["moves"]):
            raise ValueError("root move dependency changed")
        best = state["best"]
        self.last_work = len(supplied) + 1
        return (best,) + tuple(move for move in supplied if move != best)

    def commit(self, key, depth, score, move):
        state = self._active(key)
        if state["phase"] != "active":
            raise RuntimeError("cannot commit after stop")
        if not isinstance(depth, int) or depth <= state["depth"]:
            raise ValueError("depth must increase")
        if move not in state["moves"]:
            raise ValueError("committed move is illegal")
        if not isinstance(score, (int, float)) or not math.isfinite(score):
            raise ValueError("score must be finite")
        state.update(best=move, depth=depth, score=score)
        return ("commit", depth, score, move)

    def stop(self, key, reason):
        state = self._active(key)
        if state["phase"] != "active":
            raise RuntimeError("search already stopped")
        if reason not in ("timeout", "cancel", "complete"):
            raise ValueError("invalid stop reason")
        state["phase"] = "stopped"
        return ("stop", reason, state["best"])

    def finish(self, key):
        state = self._active(key)
        if state["phase"] != "stopped":
            raise RuntimeError("finish requires stop")
        if state["best"] not in state["moves"]:
            raise AssertionError("downstream legality violated")
        state["phase"] = "closed"
        return ("finish", state["best"], state["depth"])


class AnchorThinkSeam:
    """Bounded surrogate for Engine.think -> root -> UCI bestmove."""

    def __init__(self, ledger):
        self.ledger = ledger

    def run(self, key, legal_moves, completed_roots, interrupt_before=None):
        moves = tuple(legal_moves)
        trace = [self.ledger.begin(key, moves, moves[0])]

        for depth, score, move in completed_roots:
            self.ledger.order(key, moves)
            if depth == interrupt_before:
                trace.append(self.ledger.stop(key, "timeout"))
                trace.append(self.ledger.finish(key))
                return trace
            trace.append(self.ledger.commit(key, depth, score, move))

        trace.append(self.ledger.stop(key, "complete"))
        trace.append(self.ledger.finish(key))
        return trace

        ledger.finish("p"),
    ]
    assert traces["predicted_effect"][-1][1] == "b"

        ledger.finish("n"),
```
