from dataclasses import dataclass
from typing import Iterable, List, Optional, Sequence, Tuple


def king_capture_gate(board: Sequence[str], side: int,
                      moves: Iterable[int]) -> List[int]:
    """Primitive: filter anchor-encoded moves that capture the enemy king."""
    if len(board) != 64 or side not in (0, 1):
        raise ValueError("invalid anchor state")
    enemy_king = "k" if side == 0 else "K"
    result = []
    for move in moves:
        if type(move) is not int:
            raise TypeError("moves must be encoded integers")
        destination = (move >> 6) & 63
        if board[destination] != enemy_king:
            result.append(move)
    return result


@dataclass(frozen=True)
class AnchorPosition:
    # Mirrors Position(b, side, cast, ep, half, full).
    b: Tuple[str, ...]
    side: int
    cast: int
    ep: int
    half: int
    full: int

    def ordinary_transition(self, move: int) -> "AnchorPosition":
        src, dst = move & 63, (move >> 6) & 63
        board = list(self.b)
        piece = board[src]
        board[src], board[dst] = ".", piece
        return AnchorPosition(
            tuple(board), self.side ^ 1, self.cast, -1,
            0 if piece.lower() == "p" else self.half + 1,
            self.full + (1 if self.side else 0),
        )


class AnchorSeamSurrogate:
    # Provenance: pseudo() -> legal() -> search()/root() consumer.
    def __init__(self):
        self.position: Optional[AnchorPosition] = None
        self.epoch = 0

    def load(self, position: AnchorPosition) -> None:
        self.position = position
        self.epoch += 1

    def run_seam(self, pseudo_moves: Iterable[int]):
        if self.position is None:
            raise RuntimeError("position not loaded")
        # Gate is placed between pseudo() and make() in legal().
        legal_moves = king_capture_gate(
            self.position.b, self.position.side, pseudo_moves
        )
        # Downstream anchor obligation: mutable ordered encoded move list.
        downstream = list(legal_moves)
        return (downstream[0] if downstream else None), downstream


def enc(src: int, dst: int) -> int:
    return src | (dst << 6)


def board_with(*pieces):
    board = ["."] * 64
    for square, piece in pieces:
        board[square] = piece
    return tuple(board)
